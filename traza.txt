(a) foldr (-) 0 d hasta llegar a un número.

foldr (-) 0 [0, 1, 0, 9]
= 0 - (foldr (-) 0 [1, 0, 9])
= 0 - (1 - (foldr (-) 0 [0, 9]))
= 0 - (1 - (0 - (foldr (-) 0 [9])))
= 0 - (1 - (0 - (9 - (foldr (-) 0 []))))
= 0 - (1 - (0 - (9 - 0)))
= 0 - (1 - (0 - 9))
= 0 - (1 - (-9))
= 0 - 10
= -10



(b) foldl (-) 0 d hasta llegar a un número.

foldl (-) 0 [0, 1, 0, 9]
= foldl (-) (0 - 0) [1, 0, 9]
= foldl (-) ((0 - 0) - 1) [0, 9]
= foldl (-) (((0 - 0) - 1) - 0) [9]
= foldl (-) ((((0 - 0) - 1) - 0) - 9) []
= (((0 - 0) - 1) - 0) - 9
= ((0 - 1) - 0) - 9
= (-1 - 0) - 9
= -1 - 9
= -10
